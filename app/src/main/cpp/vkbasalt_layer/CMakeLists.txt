# Builds the real vkBasalt (https://github.com/DadSchoorse/vkBasalt, git submodule at
# app/src/main/cpp/vkBasalt) as an Android Vulkan layer: libVkLayer_vkBasalt.so.
#
# The Android Vulkan loader loads layers packaged in the APK (lib/<abi>/libVkLayer_*.so) when their
# name is passed in VkInstanceCreateInfo::ppEnabledLayerNames, see VulkanContext in melonDS-android-lib.
#
# Differences from the upstream meson build:
#   * no X11 (VKBASALT_X11=0): the on/off hotkey is unavailable, effects are always active
#   * SPIR-V headers of the built-in effects (cas, dls, fxaa, smaa, deband, lut) are produced with
#     glslc from the NDK (shader-tools) or glslangValidator, the same way meson does it upstream
#   * extra exports required by the Android loader (android_layer_exports.cpp)

if (NOT DEFINED VKBASALT_SRC_ROOT)
    set(VKBASALT_SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vkBasalt")
endif()

if (NOT EXISTS "${VKBASALT_SRC_ROOT}/src/basalt.cpp")
    message(WARNING "vkBasalt sources not found at ${VKBASALT_SRC_ROOT}. Run: git submodule update --init app/src/main/cpp/vkBasalt. The vkBasalt layer will NOT be built.")
    return()
endif()

set(VKBASALT_SRC "${VKBASALT_SRC_ROOT}/src")
set(VKBASALT_GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${VKBASALT_GEN}")

# ---- config_paths.hpp (generated by meson upstream)
file(WRITE "${VKBASALT_GEN}/config_paths.hpp"
        "#pragma once\n#define SYSCONFDIR \"/system/etc\"\n#define DATADIR \"/system/etc\"\n")

# ---- SPIR-V headers for the built-in effects
set(VKBASALT_SHADERS
        cas.frag deband.frag dls.frag full_screen_triangle.vert fxaa.frag lut.frag
        smaa_blend.frag smaa_blend.vert smaa_edge_color.frag smaa_edge_luma.frag
        smaa_edge.vert smaa_neighbor.frag smaa_neighbor.vert)

set(VKBASALT_GLSLC "")
if (DEFINED ANDROID_NDK)
    file(GLOB _ndk_glslc "${ANDROID_NDK}/shader-tools/*/glslc" "${ANDROID_NDK}/shader-tools/*/glslc.exe")
    if (_ndk_glslc)
        list(GET _ndk_glslc 0 VKBASALT_GLSLC)
    endif()
endif()
if (NOT VKBASALT_GLSLC)
    find_program(VKBASALT_GLSLC NAMES glslc)
endif()
find_program(VKBASALT_GLSLANG NAMES glslangValidator)

if (NOT VKBASALT_GLSLC AND NOT VKBASALT_GLSLANG)
    message(FATAL_ERROR "vkBasalt needs glslc (NDK shader-tools) or glslangValidator to compile its built-in effect shaders")
endif()

set(VKBASALT_SHADER_HEADERS "")
foreach (shader ${VKBASALT_SHADERS})
    string(REGEX MATCH "\\.(frag|vert)$" _unused "${shader}")
    set(_stage "${CMAKE_MATCH_1}")
    set(_in "${VKBASALT_SRC}/shader/${shader}.glsl")
    set(_out "${VKBASALT_GEN}/${shader}.h")
    if (VKBASALT_GLSLC)
        add_custom_command(OUTPUT "${_out}"
                COMMAND "${VKBASALT_GLSLC}" -fshader-stage=${_stage} -mfmt=num -I "${VKBASALT_SRC}/shader" -o "${_out}" "${_in}"
                DEPENDS "${_in}" VERBATIM)
    else()
        add_custom_command(OUTPUT "${_out}"
                COMMAND "${VKBASALT_GLSLANG}" -V -x -I${VKBASALT_SRC}/shader -S ${_stage} "${_in}" -o "${_out}"
                DEPENDS "${_in}" VERBATIM)
    endif()
    list(APPEND VKBASALT_SHADER_HEADERS "${_out}")
endforeach()

# ---- reshadefx front-end (static)
set(VKBASALT_RESHADE_SOURCES
        effect_codegen_spirv.cpp effect_expression.cpp effect_lexer.cpp
        effect_parser.cpp effect_preprocessor.cpp effect_symbol_table.cpp)
list(TRANSFORM VKBASALT_RESHADE_SOURCES PREPEND "${VKBASALT_SRC}/reshade/")

add_library(vkbasalt_reshade STATIC ${VKBASALT_RESHADE_SOURCES})
target_include_directories(vkbasalt_reshade PRIVATE
        "${VKBASALT_SRC}"
        "${VKBASALT_SRC}/reshade"                  # spirv.hpp shim
        "${CMAKE_CURRENT_SOURCE_DIR}/include")      # spirv/unified1/* (KhronosGroup/SPIRV-Headers, MIT)
target_compile_options(vkbasalt_reshade PRIVATE -Wno-unknown-pragmas -fvisibility=hidden -fPIC)
target_compile_features(vkbasalt_reshade PRIVATE cxx_std_20)

# ---- layer
set(VKBASALT_SOURCES
        basalt.cpp buffer.cpp command_buffer.cpp config.cpp descriptor_set.cpp effect_cas.cpp effect.cpp
        effect_deband.cpp effect_dls.cpp effect_fxaa.cpp effect_lut.cpp effect_reshade.cpp effect_simple.cpp
        effect_smaa.cpp effect_transfer.cpp fake_swapchain.cpp format.cpp framebuffer.cpp graphics_pipeline.cpp
        image.cpp image_view.cpp keyboard_input.cpp logger.cpp logical_swapchain.cpp lut_cube.cpp memory.cpp
        renderpass.cpp reshade_uniforms.cpp sampler.cpp shader.cpp stb_image.c stb_image_resize.c util.cpp
        vkdispatch.cpp)
list(TRANSFORM VKBASALT_SOURCES PREPEND "${VKBASALT_SRC}/")

add_library(VkLayer_vkBasalt SHARED
        ${VKBASALT_SOURCES}
        "${CMAKE_CURRENT_SOURCE_DIR}/android_layer_exports.cpp"
        ${VKBASALT_SHADER_HEADERS})

target_include_directories(VkLayer_vkBasalt PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/include"   # vulkan/vk_layer.h (not shipped with every NDK)
        "${VKBASALT_SRC}"
        "${VKBASALT_SRC}/reshade"
        "${VKBASALT_GEN}")
target_compile_definitions(VkLayer_vkBasalt PRIVATE VKBASALT_X11=0)
target_compile_options(VkLayer_vkBasalt PRIVATE -Wno-unknown-pragmas -fvisibility=hidden -fPIC)
target_compile_features(VkLayer_vkBasalt PRIVATE cxx_std_20)
set_target_properties(VkLayer_vkBasalt PROPERTIES OUTPUT_NAME VkLayer_vkBasalt C_STANDARD 11)
target_link_libraries(VkLayer_vkBasalt PRIVATE vkbasalt_reshade ${CMAKE_DL_LIBS})
if (ANDROID)
    target_link_libraries(VkLayer_vkBasalt PRIVATE log)
    target_link_options(VkLayer_vkBasalt PRIVATE -Wl,-z,max-page-size=16384)
endif()
